board = [
    [1, 0, 0, 4],
    [0, 4, 1, 0],
    [0, 1, 4, 0],
    [4, 0, 0, 1]
]

print("Original Sudoku:")
for row in board:
    print(row)

def valid(r, c, n):
    for i in range(4):
        if board[r][i] == n or board[i][c] == n:
            return False

    for i in range(2):
        for j in range(2):
            if board[r//2*2+i][c//2*2+j] == n:
                return False

    return True

def solve():
    for r in range(4):
        for c in range(4):
            if board[r][c] == 0:
                for n in range(1, 5):
                    if valid(r, c, n):
                        board[r][c] = n

                        if solve():
                            return True

                        board[r][c] = 0

                return False

    return True

if solve():
    print("\nSolved Sudoku:")
    for row in board:
        print(row)
else:
    print("No Solution!")
