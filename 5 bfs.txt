# to implement BFS to solve robot path plnning problem in python using grids containing/ with obstacles#

grid = [
    [0,0,1,0],
    [0,0,1,0],
    [1,0,0,0],
    [0,0,0,0]
]
start = (0,0)
goal = (3,3)

queue = [[start]]
visited = [start]

while queue:
    path = queue.pop(0)
    x,y = path[-1]

    if (x,y) == goal:
        print("path found")
        for position in path:
            print(position)
        print("total moves",len(path)-1)
        break

    moves=[(1,0),(-1,0),(0,1),(0,-1)]

    for dx,dy in moves:
        nx = x+dx
        ny = y+dy

        if 0 <= nx <4 and 0<= ny <4 :
            if grid [nx][ny]==0 and (nx,ny) not in visited:
                visited.append((nx,ny))
                queue.append(path + [(nx,ny)])
else:
    print("no path found")
